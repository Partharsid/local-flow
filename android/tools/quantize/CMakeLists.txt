cmake_minimum_required(VERSION 3.22)
project(wquantize C CXX)
set(CMAKE_CXX_STANDARD 17)
set(TP ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16+i8mm" CACHE STRING "" FORCE)
add_subdirectory(${TP}/llama.cpp llama_build EXCLUDE_FROM_ALL)
set(EX ${TP}/whisper.cpp/examples)
add_executable(wquantize ${EX}/quantize/quantize.cpp ${EX}/common-ggml.cpp)
target_include_directories(wquantize PRIVATE ${EX} ${TP}/whisper.cpp/include ${TP}/llama.cpp/ggml/include)
target_link_libraries(wquantize PRIVATE ggml)
